# Pinned dependencies for create_bsky_post.py.
#
# Versions are pinned exactly for reproducibility and supply-chain hardening:
# this is a security-sensitive tool, so an unexpected transitive upgrade should
# be a deliberate, reviewed change rather than something that happens silently.
#
# requests / beautifulsoup4 / pillow are used directly. urllib3 and idna are
# transitive dependencies of requests, but the SSRF/TLS pinning in
# _PinnedHTTPSAdapter relies on urllib3 >= 2 behaviour and _ascii_hostname uses
# the idna package directly, so both are pinned explicitly here. urllib3 2.7.0
# includes streaming decompression safeguards; the decoder allowlist excludes
# optional content codings whose installed backend may allocate without bounds.
# These pins require Python 3.10 or newer.
requests==2.32.3
beautifulsoup4==4.13.4
pillow==12.3.0
urllib3==2.7.0
idna==3.10
